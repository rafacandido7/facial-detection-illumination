MODULE DOCSTRING at file top:
  """
  Build LFW subset: select 25 images/condition from lfw-deepfunneled via lfw_attributes.txt,
  copy to dataset/lfw_subset/, auto-annotate GT bboxes via Haar cascade.

  Paper methodology note (Section III / Section V):
    GT bounding boxes for the LFW subset were generated via Haar cascade
    (haarcascade_frontalface_default.xml), consistent with the lfw-deepfunneled creation
    procedure which used Viola-Jones (Haar) centering. Haar F1 scores on this subset
    therefore represent an upper bound relative to the other detectors.
  """

CONSTANTS (exact):
  PROJECT_ROOT = Path(__file__).resolve().parent.parent
  ATTR_URL = "https://www.cs.columbia.edu/CAVE/databases/pubfig/download/lfw_attributes.txt"
  ATTR_LOCAL = PROJECT_ROOT / "dataset/lfw/lfw_attributes.txt"
  LFW_DIR = PROJECT_ROOT / "dataset/lfw/lfw-deepfunneled/lfw-deepfunneled"
  OUT_DIR = PROJECT_ROOT / "dataset/lfw_subset"
  FALLBACK_BBOX = (63, 63, 187, 187)
  N_PER_CONDITION = 25
  COL_PERSON = 1; COL_IMG = 2
  COL_HARSH = 24; COL_FLASH = 25; COL_SOFT = 26; COL_OUTDOOR = 27

CONDITIONS dict — copy from RESEARCH.md skeleton exactly:
  "bright":      score=soft+outdoor,     filter=soft>1.0 and outdoor>0
  "dark":        score=-(harsh+soft),    filter=harsh<-1.0 and soft<-0.5
  "overexposed": score=flash,            filter=flash>1.5
  "lateral":     score=harsh,            filter=harsh>0.8 and flash<0.5

main() steps:
  1. Download ATTR_LOCAL if missing via urllib.request.urlretrieve(ATTR_URL, ATTR_LOCAL)
  2. Parse: data_lines = [l.split('\t') for l in open(ATTR_LOCAL) if not l.startswith('#') and l.strip()][1:]
  3. Init Haar ONCE: cascade = cv2.CascadeClassifier(cv2.data.haarcascades + 'haarcascade_frontalface_default.xml')
  4. gt_entries = []
  5. For each cond:
       - mkdir OUT_DIR/cond
       - Build candidates: filter rows, build src path as LFW_DIR/name/f"{name}_{imgnum:04d}.jpg",
         check src.exists(), append (score, src). Apply .replace(' ', '_') to name.
       - Sort desc, take first N_PER_CONDITION
       - For each src: shutil.copy2(src, dst); Haar-detect on dst;
         if len(faces)>0: bbox from faces[0]; else: print WARNING + use FALLBACK_BBOX
       - rel = str(dst.resolve().relative_to(PROJECT_ROOT))
       - gt_entries.append({"file": rel, "condition": cond, "faces": [bbox]})
       - print(f"{cond}: N images copied")
  6. Write gt.json: {"dataset": "lfw_subset", "iou_threshold": 0.5, "images": gt_entries}
  7. Self-check: assert gt_entries[0]["file"] does not start with '/' or './'
     print sample key for manual verification

IMPORTS: import cv2, json, shutil, urllib.request; from pathlib import Path
ENTRYPOINT: if __name__ == "__main__": main()
